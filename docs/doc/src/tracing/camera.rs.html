1//! A module for camera simulation in a ray tracing environment.
2//!
3//! This module provides the `Camera` struct, which is responsible for generating
4//! rays that are cast into the scene. It simulates a thin-lens camera model,
5//! allowing for features like depth of field and configurable field of view.
6
7use crate::ray::Ray;
8use crate::vec;
9use crate::vec::{Point3, Vec3};
10
11/// Represents a virtual camera for generating rays in the scene.
12///
13/// The camera is defined by its position, orientation, and lens properties.
14/// It creates rays that originate from the camera's lens and travel through
15/// a virtual viewport, simulating a pinhole or thin-lens camera model.
16pub struct Camera {
17    /// The origin point of the camera, representing the center of the lens.
18    origin: Point3,
19    /// The bottom-left corner of the virtual viewport in world space.
20    lower_left_corner: Point3,
21    /// A vector representing the width of the viewport.
22    horizontal: Vec3,
23    /// A vector representing the height of the viewport.
24    vertical: Vec3,
25    /// The orthonormal basis vector for the camera's horizontal orientation.
26    u: Vec3,
27    /// The orthonormal basis vector for the camera's vertical orientation.
28    v: Vec3,
29    /// The radius of the camera's thin lens, used for simulating depth of field.
30    lens_radius: f32,
31}
32
33impl Camera {
34    /// Creates and configures a new `Camera`.
35    ///
36    /// This constructor sets up the camera's geometry based on its position,
37    /// the point it's looking at, and its field of view. It also configures
38    /// the lens for depth-of-field effects.
39    ///
40    /// # Arguments
41    ///
42    /// * `from` - The position of the camera in world space (`Point3`).
43    /// * `lookat` - The point in world space that the camera is looking at (`Point3`).
44    /// * `vup` - The "view up" vector, defining the camera's upward orientation (e.g., `Vec3::new(0.0, 1.0, 0.0)`).
45    /// * `vfov` - The vertical field-of-view, in degrees.
46    /// * `aspect_ratio` - The aspect ratio of the viewport (width / height).
47    /// * `aperture` - The diameter of the camera's aperture. A larger aperture creates a more pronounced depth-of-field effect.
48    /// * `focus_dist` - The distance from the camera to the focal plane. Objects at this distance will be perfectly in focus.
49    ///
50    /// # Returns
51    ///
52    /// A new `Camera` instance configured with the specified parameters.
53    pub fn new(
54        from: Point3,
55        lookat: Point3,
56        vup: Vec3,
57        vfov: f32, // vertical field-of-view in degrees
58        aspect_ratio: f32,
59        aperture: f32,
60        focus_dist: f32,
61    ) -> Self {
62        // Calculate viewport dimensions from the vertical FOV
63        let theta = vfov.to_radians();
64        let h = f32::tan(theta / 2.0);
65        let viewport_height = 2.0 * h;
66        let viewport_width = aspect_ratio * viewport_height;
67
68        // Create an orthonormal basis (u, v, w) for the camera's orientation
69        // w is the backward-pointing vector (from lookat to camera)
70        let w = vec::unit_vector(from - lookat);
71        // u is the side-pointing vector, found via cross product of up vector and w
72        let u = vec::unit_vector(vec::cross(vup, w));
73        // v is the new "up" vector, perpendicular to w and u
74        let v = vec::cross(w, u);
75
76        let origin = from;
77        // The horizontal and vertical vectors span the focal plane
78        let horizontal = focus_dist * viewport_width * u;
79        let vertical = focus_dist * viewport_height * v;
80        // The lower-left corner of the focal plane
81        let lower_left_corner = origin - horizontal / 2.0 - vertical / 2.0 - focus_dist * w;
82
83        let lens_radius = aperture / 2.0;
84
85        Self {
86            origin,
87            lower_left_corner,
88            horizontal,
89            vertical,
90            u,
91            v,
92            lens_radius,
93        }
94    }
95
96    /// Generates a ray from the camera through a point on its virtual viewport.
97    ///
98    /// This function simulates a thin lens by originating the ray from a random
99    /// point on the camera's aperture. The direction is set to a point on the
100    /// focal plane, creating a depth-of-field effect.
101    ///
102    /// # Arguments
103    ///
104    /// * `s` - The horizontal coordinate on the viewport, in the range `[0, 1]`.
105    /// * `t` - The vertical coordinate on the viewport, in the range `[0, 1]`.
106    ///
107    /// # Returns
108    ///
109    /// A `Ray` that originates from the camera's lens and travels into the scene.
110    pub fn get_ray(&self, s: f32, t: f32) -> Ray {
111        // Find a random point on the lens disk to simulate a thin lens
112        let rd = self.lens_radius * vec::random_in_unit_disk();
113        // Calculate the offset on the lens from the camera's origin
114        let offset = self.u * rd.x() + self.v * rd.y();
115
116        Ray::new(
117            // Ray origin: camera origin + lens offset
118            self.origin + offset,
119            // Ray direction: vector from the lens point to the focal plane point
120            self.lower_left_corner + s * self.horizontal + t * self.vertical - self.origin - offset,
121        )
122    }
123}
