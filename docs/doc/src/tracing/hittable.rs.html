1//! Defines the core components for ray-object intersection in the ray tracer.
2//!
3//! This module provides the `Hittable` trait, which is a fundamental abstraction for any
4//! object in the scene that can be intersected by a ray (e.g., spheres, planes). It also
5//! provides the `HitRecord` struct, which stores detailed information about such an
6//! intersection, including the point of contact, surface normal, and material.
7
8use std::sync::Arc;
9
10use crate::material::Material;
11use crate::ray::Ray;
12use crate::vec::{self, Point3, Vec3};
13
14/// A struct to store information about a ray-object intersection.
15///
16/// When a ray hits a `Hittable` object, a `HitRecord` is populated with details
17/// about the hit, such as the intersection point, the surface normal, and the
18/// material of the object.
19#[derive(Clone, Default)]
20pub struct HitRecord {
21    /// The point in 3D space where the intersection occurred.
22    pub p: Point3,
23    /// The surface normal vector at the point of intersection.
24    /// This normal always points against the incident ray.
25    pub normal: Vec3,
26    /// A reference to the material of the object that was hit.
27    pub mat: Option<Arc<dyn Material>>,
28    /// The parameter `t` along the ray where the intersection occurred, such that `p = ray.origin() + t * ray.direction()`.
29    pub t: f32,
30    /// A boolean indicating whether the ray hit the front face of the surface.
31    /// `true` if the ray strikes the object from the outside, `false` otherwise.
32    pub front_face: bool,
33}
34
35impl HitRecord {
36    /// Creates a new, default `HitRecord`.
37    pub fn new() -> Self {
38        Self::default()
39    }
40
41    /// Sets the surface normal based on the ray's direction.
42    ///
43    /// This method ensures the normal vector in the `HitRecord` always points
44    /// against the incoming ray. It determines whether the ray hit the front or
45    /// back face of the surface and flips the `outward_normal` if necessary.
46    ///
47    /// # Arguments
48    ///
49    /// * `r` - The ray that intersected the surface.
50    /// * `outward_normal` - The geometric normal of the surface, assuming it points "outward".
51    ///                      This vector is assumed to have unit length.
52    pub fn set_face_normal(&mut self, r: &Ray, outward_normal: Vec3) {
53        // The dot product tells us if the ray and outward_normal are in opposite directions.
54        // If dot < 0, the ray is hitting the front face.
55        self.front_face = vec::dot(r.direction(), outward_normal) < 0.0;
56        self.normal = if self.front_face {
57            outward_normal
58        } else {
59            // Ray is inside the object; reverse the normal so it points against the ray.
60            -outward_normal
61        };
62    }
63}
64
65/// A trait for objects that can be intersected by a ray.
66///
67/// Any object in the scene, such as a sphere or a collection of objects,
68/// must implement this trait to be rendered. The `Send + Sync` bounds are
69/// required to allow for safe multi-threaded rendering.
70pub trait Hittable: Send + Sync {
71    /// Determines if a ray intersects with the object.
72    ///
73    /// # Arguments
74    ///
75    /// * `ray` - The ray to test for intersection.
76    /// * `t_min` - The minimum valid value for the ray parameter `t`.
77    /// * `t_max` - The maximum valid value for the ray parameter `t`.
78    /// * `rec` - A mutable `HitRecord` to be populated with intersection data if a hit occurs.
79    ///
80    /// # Returns
81    ///
82    /// `true` if the ray hits the object within the interval `[t_min, t_max]`,
83    /// and `rec` is updated. Otherwise, returns `false`.
84    fn hit(&self, ray: &Ray, t_min: f32, t_max: f32, rec: &mut HitRecord) -> bool;
85}
