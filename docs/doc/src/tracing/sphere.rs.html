1//! A module providing a `Sphere` geometric primitive for the ray tracer.
2//!
3//! This module defines the `Sphere` struct, which represents a sphere in 3D space,
4//! and implements the `Hittable` trait for it. This allows rays to intersect
5//! with spheres in the scene.
6
7use std::sync::Arc;
8
9use crate::hittable::{HitRecord, Hittable};
10use crate::material::Material;
11use crate::ray::Ray;
12use crate::vec::{self, Point3};
13
14/// Represents a sphere object in 3D space.
15///
16/// A `Sphere` is defined by its center point, a radius, and the material
17/// that determines its appearance and how it interacts with light.
18pub struct Sphere {
19    /// The center point of the sphere in world coordinates.
20    center: Point3,
21    /// The radius of the sphere.
22    radius: f32,
23    /// A reference to the material of the sphere's surface.
24    mat: Arc<dyn Material>,
25}
26
27impl Sphere {
28    /// Creates a new `Sphere`.
29    ///
30    /// # Arguments
31    ///
32    /// * `cen` - The center `Point3` of the sphere.
33    /// * `r` - The `f32` radius of the sphere.
34    /// * `material` - An `Arc<dyn Material>` that defines the sphere's surface properties.
35    ///
36    /// # Returns
37    ///
38    /// A new `Sphere` instance.
39    pub fn new(cen: Point3, r: f32, material: Arc<dyn Material>) -> Self {
40        Self {
41            center: cen,
42            radius: r,
43            mat: material,
44        }
45    }
46}
47
48impl Hittable for Sphere {
49    /// Determines if a ray intersects with the sphere.
50    ///
51    /// This method calculates the intersection by solving the quadratic equation
52    /// that results from the formula of a ray `P(t) = O + t*D` and the formula of a
53    /// sphere `(P - C) · (P - C) = r^2`.
54    ///
55    /// # Arguments
56    ///
57    /// * `r` - The ray to test for intersection.
58    /// * `t_min` - The minimum valid value for the ray parameter `t`.
59    /// * `t_max` - The maximum valid value for the ray parameter `t`.
60    /// * `rec` - A mutable `HitRecord` that will be populated with intersection data if a hit occurs.
61    ///
62    /// # Returns
63    ///
64    /// `true` if the ray hits the sphere within the interval `[t_min, t_max]`, and `rec`
65    /// is updated with the hit details. Otherwise, returns `false`.
66    fn hit(&self, r: &Ray, t_min: f32, t_max: f32, rec: &mut HitRecord) -> bool {
67        // Deriving the quadratic equation for ray-sphere intersection:
68        // (ray_origin + t*ray_direction - sphere_center) · (ray_origin + t*ray_direction - sphere_center) = radius^2
69        // Let oc = ray_origin - sphere_center.
70        // (t*ray_direction + oc) · (t*ray_direction + oc) = radius^2
71        // t^2*(D·D) + 2*t*(D·oc) + (oc·oc) - r^2 = 0
72        // Where D is ray_direction.
73        // This is a quadratic equation at^2 + bt + c = 0, with:
74        // a = D·D
75        // b = 2*(D·oc)
76        // c = (oc·oc) - r^2
77        // We use a simplified form with half_b = D·oc.
78        let oc = r.origin() - self.center;
79        let a = r.direction().length_squared();
80        let half_b = vec::dot(oc, r.direction());
81        let c = oc.length_squared() - self.radius * self.radius;
82        let discriminant = half_b * half_b - a * c;
83
84        // If the discriminant is negative, there are no real roots, so no intersection.
85        if discriminant < 0.0 {
86            return false;
87        }
88
89        let sqrt_d = discriminant.sqrt();
90
91        // Find the nearest root that lies in the acceptable range [t_min, t_max].
92        // Check the smaller root first.
93        let mut root = (-half_b - sqrt_d) / a;
94        if root <= t_min || t_max <= root {
95            // Smaller root is outside the range, check the larger root.
96            root = (-half_b + sqrt_d) / a;
97            if root <= t_min || t_max <= root {
98                // Both roots are outside the range.
99                return false;
100            }
101        }
102
103        // A valid intersection was found. Populate the HitRecord.
104        rec.t = root;
105        rec.p = r.at(rec.t);
106        let outward_normal = (rec.p - self.center) / self.radius;
107        rec.set_face_normal(r, outward_normal);
108        rec.mat = Some(self.mat.clone());
109
110        true
111    }
112}
