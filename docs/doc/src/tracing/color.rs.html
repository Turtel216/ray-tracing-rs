1//! A module for color representation and manipulation.
2//!
3//! This module defines the `Color` type, an alias for `Vec3`, to represent
4//! RGB color values. It also provides utility functions for processing these
5//! _color!s for output, such as applying gamma correction and converting them
6//! into a format suitable for image files like PPM.
7
8use crate::util;
9use crate::vec::Vec3;
10
11/// A type alias for `Vec3` to represent an RGB color.
12///
13/// The `x`, `y`, and `z` components of the `Vec3` correspond to the
14/// red, green, and blue channels of the color, respectively. These values are
15/// typically represented as floating-point numbers in the range `[0.0, 1.0]`.
16pub type Color = Vec3;
17
18/// Converts a floating-point `Color` value into a byte vector for PPM image output.
19///
20/// This function performs several key steps:
21/// 1.  **Averaging:** It scales the raw accumulated color by the number of samples
22///     taken for that pixel to get the average color.
23/// 2.  **Gamma Correction:** It applies a gamma-2 correction (by taking the square root)
24///     to the color components. This is crucial for accurate color reproduction on most displays.
25/// 3.  **Scaling and Clamping:** It maps the floating-point color components from the `[0.0, 1.0]`
26///     range to the integer range `[0, 255]`.
27///
28/// The final output is a `Vec<u8>` containing a string in the format "R G B\n".
29///
30/// # Arguments
31///
32/// * `pixel_color` - The accumulated `Color` value from all samples for a single pixel.
33/// * `samples_per_pixel` - The total number of samples that contributed to `pixel_color`.
34///
35/// # Returns
36///
37/// A `Vec<u8>` representing the final, formatted RGB values for a single pixel.
38///
39/// # Examples
40///
41/// ```
42/// use crate::color::{Color, write_color};
43///
44/// // A sample color (e.g., a medium gray) accumulated over 100 samples.
45/// // In a real scenario, this would be the sum of 100 different color values.
46/// let accumulated_color = Color::new(50.0, 50.0, 50.0);
47/// let samples = 100;
48///
49/// let byte_color = write_color(accumulated_color, samples);
50///
51/// // The expected process:
52/// // 1. Average: 50.0 / 100 = 0.5
53/// // 2. Gamma correct: sqrt(0.5) ≈ 0.707
54/// // 3. Scale and clamp: 256 * 0.707 ≈ 181
55/// // 4. Format: "181 181 181\n"
56/// assert_eq!(byte_color, b"181 181 181\n");
57/// ```
58pub fn write_color(pixel_color: Color, samples_per_pixel: i32) -> Vec<u8> {
59    let mut r = pixel_color.x();
60    let mut g = pixel_color.y();
61    let mut b = pixel_color.z();
62
63    // Divide the color by the number of samples to get the average color.
64    // This is part of antialiasing.
65    let scale = 1.0 / samples_per_pixel as f32;
66    r *= scale;
67    g *= scale;
68    b *= scale;
69
70    // Apply gamma correction (gamma = 2.0).
71    // This transforms the color from linear space to gamma space.
72    r = r.sqrt();
73    g = g.sqrt();
74    b = b.sqrt();
75
76    // Format the color components into a string "R G B\n" and convert to bytes.
77    // The values are scaled to [0, 255] and clamped to ensure they are within a valid range.
78    format!(
79        "{} {} {}\n",
80        (256.0 * util::clamp(r, 0.0, 0.999)) as u8,
81        (256.0 * util::clamp(g, 0.0, 0.999)) as u8,
82        (256.0 * util::clamp(b, 0.0, 0.999)) as u8,
83    )
84    .into_bytes()
85}
