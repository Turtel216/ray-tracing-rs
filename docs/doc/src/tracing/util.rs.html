1//! A collection of utility functions for math and random number generation.
2//!
3//! This module provides simple, commonly used helper functions for tasks
4//! such as generating random floating-point numbers and clamping values
5//! within a specified range.
6use rand::Rng;
7
8/// Generates a random `f32` value in the range `[0.0, 1.0)`.
9///
10/// This function is a convenient wrapper around `rand::thread_rng().gen()`.
11///
12/// # Examples
13///
14/// ```
15/// let random_val = random_double();
16/// assert!(random_val >= 0.0 && random_val < 1.0);
17/// ```
18#[inline]
19pub fn random_double() -> f32 {
20    rand::rng().random()
21}
22
23/// Generates a random `f32` value within a specified range `[min, max)`.
24///
25/// # Arguments
26///
27/// * `min` - The inclusive lower bound of the random number.
28/// * `max` - The exclusive upper bound of the random number.
29///
30/// # Examples
31///
32/// ```
33/// let random_val = random_double_range(5.0, 10.0);
34/// assert!(random_val >= 5.0 && random_val < 10.0);
35/// ```
36#[inline]
37pub fn random_double_range(min: f32, max: f32) -> f32 {
38    // This uses the formula: min + (max - min) * random_double()
39    // It is written this way to potentially leverage the Fused Multiply-Add (FMA) CPU instruction
40    // for a slight performance improvement, though compiler optimizations may vary.
41    (max - min).mul_add(random_double(), min)
42}
43
44/// Clamps a floating-point number `x` to be within the range `[min, max]`.
45///
46/// If `x` is less than `min`, it returns `min`. If `x` is greater than `max`,
47/// it returns `max`. Otherwise, it returns `x`.
48///
49/// Note: This function provides the same functionality as the built-in `f32::clamp`.
50///
51/// # Arguments
52///
53/// * `x` - The value to clamp.
54/// * `min` - The inclusive lower bound.
55/// * `max` - The inclusive upper bound.
56///
57/// # Examples
58///
59/// ```
60/// assert_eq!(clamp(5.0, 0.0, 10.0), 5.0);
61/// assert_eq!(clamp(-5.0, 0.0, 10.0), 0.0);
62/// assert_eq!(clamp(15.0, 0.0, 10.0), 10.0);
63/// ```
64#[inline]
65pub fn clamp(x: f32, min: f32, max: f32) -> f32 {
66    if x < min {
67        min
68    } else if x > max {
69        max
70    } else {
71        x
72    }
73}
